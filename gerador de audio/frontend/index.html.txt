<script>
document.addEventListener('DOMContentLoaded', () => {
    const el = {
        generateScriptBtn: document.getElementById('generate-script-btn'),
        scriptText: document.getElementById('script-text'),
        scriptTopic: document.getElementById('script-topic'),
        imageUpload: document.getElementById('image-upload'),
        generateBtn: document.getElementById('generate-btn'),
        downloadBtn: document.getElementById('download-btn'),
        audioPlayer: document.getElementById('audio-player'),
        loader: document.getElementById('loader'),
        errorMessage: document.getElementById('error-message'),
        voiceSelect: document.getElementById('voice-select'),
        voiceTone: document.getElementById('voice-tone'),
        musicStyleSelect: document.getElementById('music-style-select'),
        musicVolumeSlider: document.getElementById('music-volume-slider'),
        musicVolumeLabel: document.getElementById('music-volume-label'),
        mixAudioCheckbox: document.getElementById('mix-audio-checkbox'),
    };

    const VOICES = { "Vozes": [{ id: "Kore", label: "Voz 1 (Firme)" }, { id: "Puck", label: "Voz 2 (Animada)" }] };
    Object.values(VOICES).flat().forEach(v => {
        const opt = document.createElement("option");
        opt.value = v.id;
        opt.textContent = v.label;
        el.voiceSelect.appendChild(opt);
    });

    let generatedAudioBlob = null;
    let uploadedImageData = null;

    function showError(msg) { el.errorMessage.textContent = msg || ''; }

    async function apiCall(endpoint, body) {
        const response = await fetch(endpoint, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(body),
        });
        if (!response.ok) {
            const err = await response.json();
            throw new Error(err.error || 'Erro desconhecido no servidor.');
        }
        return response.json();
    }
    
    // Converte Base64 para um ArrayBuffer
    function base64ToArrayBuffer(base64) {
        const binaryString = window.atob(base64);
        const len = binaryString.length;
        const bytes = new Uint8Array(len);
        for (let i = 0; i < len; i++) {
            bytes[i] = binaryString.charCodeAt(i);
        }
        return bytes.buffer;
    }

    el.generateScriptBtn.addEventListener('click', async () => {
        showError();
        let contents = [];
        if (uploadedImageData) {
            contents.push({ text: "Crie um roteiro de anúncio para esta imagem." });
            contents.push({ inlineData: { mimeType: uploadedImageData.type, data: uploadedImageData.data } });
        } else {
            const topic = el.scriptTopic.value.trim();
            if (!topic) return showError("Por favor, digite um tema.");
            contents.push({ text: `Crie um roteiro de anúncio sobre: ${topic}` });
        }
        try {
            const data = await apiCall('/api/generate-script', { contents });
            el.scriptText.value = data.script;
        } catch (err) {
            showError(err.message);
        }
    });

    el.generateBtn.addEventListener('click', async () => {
        const script = el.scriptText.value.trim();
        if (!script) return showError("Por favor, gere ou digite um roteiro.");
        
        el.loader.classList.remove('hidden');
        showError();
        el.audioPlayer.classList.add('hidden');
        el.downloadBtn.classList.add('hidden');

        try {
            const prompt = `Narre o seguinte texto com um tom ${el.voiceTone.value}: "${script}"`;
            const payload = {
                contents: [{ parts: [{ text: prompt }] }],
                generationConfig: {
                    responseModalities: ["AUDIO"],
                    speechConfig: { voiceConfig: { prebuiltVoiceConfig: { voiceName: el.voiceSelect.value } } }
                }
            };

            const data = await apiCall('/api/generate-audio', payload);
            
            const audioBuffer = base64ToArrayBuffer(data.audioData);
            generatedAudioBlob = new Blob([audioBuffer], { type: data.mimeType });
            
            el.audioPlayer.src = URL.createObjectURL(generatedAudioBlob);
            el.audioPlayer.classList.remove('hidden');
            el.downloadBtn.classList.remove('hidden');
        } catch (err) {
            showError(err.message);
        } finally {
            el.loader.classList.add('hidden');
        }
    });
    
    el.downloadBtn.addEventListener('click', () => {
         if (!generatedAudioBlob) return;
         const a = document.createElement("a");
         a.href = URL.createObjectURL(generatedAudioBlob);
         a.download = `narracao_${Date.now()}.mp3`; 
         document.body.appendChild(a);
         a.click();
         document.body.removeChild(a);
    });

    el.imageUpload.addEventListener('change', (event) => {
        const file = event.target.files[0];
        if (!file) return;
        const reader = new FileReader();
        reader.onloadend = () => {
            const base64 = reader.result.split(',')[1];
            uploadedImageData = { data: base64, type: file.type };
        };
        reader.readAsDataURL(file);
    });

    el.musicVolumeSlider.addEventListener('input', (e) => {
        el.musicVolumeLabel.textContent = e.target.value;
    });
});
</script>
